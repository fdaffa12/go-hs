            $(document).ready(function() {
                // Event handler untuk tombol import
                $('#importExcel').on('click', function() {
                    $('#importFile').click();
                });

                // Event handler untuk input file
                $('#importFile').on('change', function(event) {
                    const selectedDate = new Date($('#dateSelect').val());
                    const currentDate = new Date();

                    // Reset waktu ke 00:00:00 untuk perbandingan yang akurat
                    selectedDate.setHours(0, 0, 0, 0);
                    currentDate.setHours(0, 0, 0, 0);

                    if (selectedDate < currentDate) {
                        Swal.fire({
                            title: 'Tidak Diizinkan!',
                            text: 'Anda tidak dapat mengimpor data untuk tanggal yang sudah berlalu.',
                            icon: 'error',
                            confirmButtonColor: '#3085d6',
                            confirmButtonText: 'OK'
                        });
                        // Reset input file
                        $(this).val('');
                        return;
                    }

                    const file = event.target.files[0];
                    const reader = new FileReader();
                    reader.onload = function(e) {
                        const data = new Uint8Array(e.target.result);
                        const workbook = XLSX.read(data, {
                            type: 'array'
                        });
                        const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                        const jsonData = XLSX.utils.sheet_to_json(firstSheet, {
                            header: 1
                        });

                        // Validasi header
                        const expectedHeaders = [
                            "ID_REGISTRASI",
                            "COMPONENT/ASSEMBLY",
                            "LINE",
                            "BUYER_SHORT_NAME",
                            "STYLE_NO",
                            "NUMBER_OF_MP",
                            "START_DATE",
                            "WORKING_DAY"
                        ];

                        const headers = jsonData[0];
                        const isValidHeader = expectedHeaders.every((header, index) =>
                            headers[index] === header
                        );

                        if (!isValidHeader) {
                            Swal.fire({
                                title: 'Error!',
                                text: 'Invalid file format. Please use the correct template.',
                                icon: 'error',
                                confirmButtonColor: '#3085d6',
                                confirmButtonText: 'OK'
                            });
                            return;
                        }

                        // Cek data duplikat
                        let duplicates = [];
                        let existingData = [];
                        let uniqueData = [];

                        // Kumpulkan data yang sudah ada di tabel
                        $('#lineScheduleTable tbody tr').each(function() {
                            const row = $(this);
                            const data = {
                                line: row.find('select.line-select').val(),
                                styleNo: row.find('select.style-no-select').val(),
                                buyerShortName: row.find('select.buyer-select')
                                    .val(),
                                startDate: row.find('.start-date').val()
                            };
                            // Buat key unik untuk setiap kombinasi data
                            const dataKey =
                                `${data.line}-${data.styleNo}-${data.buyerShortName}-${data.startDate}`;
                            existingData.push({
                                key: dataKey,
                                data: data
                            });
                        });

                        // Cek setiap baris data import
                        for (let i = 1; i < jsonData.length; i++) {
                            const row = jsonData[i];
                            const importData = {
                                line: row[2],
                                styleNo: row[4],
                                buyerShortName: row[3],
                                startDate: row[6]
                            };

                            // Buat key unik untuk data yang diimpor
                            const importKey =
                                `${importData.line}-${importData.styleNo}-${importData.buyerShortName}-${importData.startDate}`;

                            // Cek apakah data sudah ada di existingData atau di uniqueData
                            const isDuplicateExisting = existingData.some(item => item.key ===
                                importKey);
                            const isDuplicateImport = uniqueData.some(item => item.key ===
                                importKey);

                            if (isDuplicateExisting || isDuplicateImport) {
                                // Hanya tambahkan ke duplicates jika belum ada data yang sama
                                if (!duplicates.some(dup => dup.key === importKey)) {
                                    duplicates.push({
                                        key: importKey,
                                        message: `Row ${i + 1}: Line ${importData.line}, Style ${importData.styleNo}, Buyer ${importData.buyerShortName}, Date ${importData.startDate}`
                                    });
                                }
                            } else {
                                // Tambahkan ke uniqueData jika belum ada duplikat
                                uniqueData.push({
                                    key: importKey,
                                    data: importData,
                                    rowIndex: i
                                });
                            }
                        }

                        if (duplicates.length > 0) {
                            Swal.fire({
                                title: 'Duplicate Data Found!',
                                html: '<div style="text-align: left; margin-top: 10px;">' +
                                    '<p>The following data already exists:</p>' +
                                    '<ul style="list-style-type: disc; margin-left: 20px;">' +
                                    duplicates.map(dup => `<li>${dup.message}</li>`)
                                    .join('') +
                                    '</ul>' +
                                    '<p>Only unique data will be imported. Do you want to continue?</p></div>',
                                icon: 'warning',
                                showCancelButton: true,
                                confirmButtonColor: '#3085d6',
                                cancelButtonColor: '#d33',
                                confirmButtonText: 'Yes, import unique data',
                                cancelButtonText: 'No, cancel',
                                width: '600px'
                            }).then((result) => {
                                if (result.isConfirmed) {
                                    // Hanya import data unik
                                    const uniqueJsonData = [jsonData[0]]; // Keep header
                                    uniqueData.forEach(item => {
                                        uniqueJsonData.push(jsonData[item
                                            .rowIndex]);
                                    });

                                    appendTableData(uniqueJsonData);
                                    Swal.fire({
                                        title: 'Imported!',
                                        text: `${uniqueData.length} unique records have been imported successfully!`,
                                        icon: 'success',
                                        confirmButtonColor: '#3085d6',
                                        confirmButtonText: 'OK'
                                    });
                                }
                            });
                        } else {
                            appendTableData(jsonData);
                            Swal.fire({
                                title: 'Imported!',
                                text: 'All data has been imported successfully!',
                                icon: 'success',
                                confirmButtonColor: '#3085d6',
                                confirmButtonText: 'OK'
                            });
                        }
                    };
                    reader.readAsArrayBuffer(file);
                });

                function appendTableData(data) {
                    const tableBody = $('#lineScheduleTable tbody');
                    const existingRows = tableBody.children().length;
                    data.forEach((row, index) => {
                        if (index === 0) return; // Skip header row
                        const rowData = {
                            ID_REGISTRASI: row[0] || '',
                            COMPONENT: row[1] || '', // Kolom baru COMPONENT/ASSEMBLY
                            LINE: row[2] || '',
                            BUYER_SHORT_NAME: row[3] || '',
                            STYLE_NO: row[4] || '',
                            NUMBER_OF_MP: row[5] || '',
                            START_DATE: row[6] || '',
                            WORKING_DAY: row[7] || ''
                        };

                        // Set COMPONENT berdasarkan LINE jika tidak ada
                        if (!rowData.COMPONENT) {
                            rowData.COMPONENT = rowData.LINE === 'COMPONENT' ? 'COMPONENT' :
                                'ASSEMBLY';
                        }

                        addRow(rowData, existingRows + index);
                    });
                }
            });

            function addRow(row, index = null, isNew = false) {
                const tableBody = $('#lineScheduleTable tbody');
                const deleteStatus = row.DELETE_STATUS !== undefined ? row.DELETE_STATUS : 0;
                const buttonClass = deleteStatus == 1 ? 'btn-danger' : 'btn-warning';
                const buttonText = deleteStatus == 1 ? '<i class="fas fa-minus-circle"></i>' :
                    '<i class="fas fa-minus-circle"></i>';

                // Tentukan nilai Component/Assembly berdasarkan nilai Line dari database
                const componentType = row.LINE === 'AREA' ? 'COMPONENT' : 'ASSEMBLY';
                const isLineDisabled = componentType === 'COMPONENT';
                const lineOpacity = isLineDisabled ? '0.5' : '1';
                const lineValue = row.LINE; // Gunakan nilai LINE langsung dari database
                const showComponent = isLineDisabled;

                // Tentukan apakah field bisa diedit berdasarkan akses user
                const isEditable = {{ in_array('edit', $userAccesses) ? 'true' : 'false' }};
                const editableAttr = (isNew || isEditable) ? 'true' : 'false';

                const newRow = `
                <tr>
                    <td><input type="checkbox" class="selectRow"></td>
                    <td>${index || tableBody.children().length + 1}</td>
                    <td class="id-reg">${row.ID_REGISTRASI || ''}</td>
                    <td>
                        <select class="form-control area-select" data-column="COMPONENT" ${!isEditable ? 'disabled' : ''}>
                            ${generateAreaOptions(componentType)}
                        </select>
                    </td>
                    <td style="opacity: ${lineOpacity}">
                        <select class="form-control line-select" data-column="LINE" ${!isEditable ? 'disabled' : ''} ${isLineDisabled ? 'disabled' : ''}>
                            ${generateLineOptions(lineValue, showComponent)}
                        </select>
                    </td>
                    <td>
                        <select class="form-control buyer-select" data-column="BUYER_SHORT_NAME" ${!isEditable ? 'disabled' : ''}>
                            ${generateBuyerOptions(row.BUYER_SHORT_NAME)}
                        </select>
                    </td>
                    <td>
                        <select class="form-control style-no-select" data-column="STYLE_NO" ${!isEditable ? 'disabled' : ''}>
                            ${generateStyleOptions(row.STYLE_NO, row.BUYER_SHORT_NAME)}
                        </select>
                    </td>
                    <td contenteditable="${editableAttr}" data-column="NUMBER_OF_MP">${row.NUMBER_OF_MP || ''}</td>
                    <td>
                        <input type="date" class="form-control start-date" value="${row.START_DATE || ''}" ${!isEditable ? 'readonly' : ''} />
                    </td>
                    <td contenteditable="false" data-column="WORKING_DAY" style="opacity: 0.5; background-color: #f5f5f5;">${row.WORKING_DAY || ''}</td>
                    <td>
                        <div class="action-btn-group">
                            @if (in_array('delete', $userAccesses))
                                <button type="button" class="btn btn-outline-secondary deleteRow delete">
                                    <i class="fas fa-trash-alt"></i>
                                </button>
                            @endif
                        </div>
                    </td>
                </tr>
            `;
                tableBody.append(newRow);

                // Pastikan nilai line select sesuai setelah row ditambahkan
                const lastRow = tableBody.find('tr:last');
                const areaSelect = lastRow.find('.area-select');
                const lineSelect = lastRow.find('.line-select');

                // Set nilai area select dan line select
                areaSelect.val(componentType);
                if (lineValue === 'COMPONENT') {
                    lineSelect.val('COMPONENT');
                } else {
                    lineSelect.val(lineValue);
                }

                updateRegistrationId();
            }


            function generateAreaOptions(selectedValue) {
                let options = '<option value="" disabled selected>-- Select Component/Assembly --</option>';
                const types = ['COMPONENT', 'ASSEMBLY'];
                types.forEach(type => {
                    options +=
                        `<option value="${type}" ${selectedValue === type ? 'selected' : ''}>${type}</option>`;
                });
                return options;
            }

            function generateLineOptions(selectedValue, showComponent = true) {
                let options = '<option value="" disabled>-- Select Line --</option>';
                if (showComponent) {
                    options += `<option value="AREA" ${selectedValue === 'AREA' ? 'selected' : ''}>AREA</option>`;
                }
                // Hanya menambahkan Line 1 sampai 30
                for (let i = 1; i <= 30; i++) {
                    options += `<option value="${i}" ${selectedValue == i ? 'selected' : ''}>LINE ${i}</option>`;
                }
                return options;
            }

            function generateStyleOptions(selectedValue, buyer) {
                let options = '<option value="" disabled selected>-- Select Style No --</option>';
                stylesData.forEach(style => {
                    if (style.BUYER_SHORT_NAME === buyer) {
                        options +=
                            `<option value="${style.STYLE_NO}" ${selectedValue == style.STYLE_NO ? 'selected' : ''}>${style.STYLE_NO}</option>`;
                    }
                });
                return options;
            }


            function generateRegistrationId(factory, component, line, styleNo, startDate) {
                const prefix = component === 'COMPONENT' ? 'A0' : 'L';
                const lineValue = component === 'COMPONENT' ? '' : line;
                return factory + '-' + prefix + lineValue + '-' + (styleNo || 'XXXX') + '-' + startDate.replace(
                    /-/g, '');
            }

            async function calculateWorkingDays(startDate) {
                try {
                    const start = new Date(startDate);
                    const selectedDate = new Date($('#dateSelect').val());

                    if (start > selectedDate) {
                        return Promise.resolve(0);
                    }

                    const response = await $.ajax({
                        url: "{{ route('holiday.getHolidayDates') }}",
                        method: 'POST',
                        data: {
                            _token: "{{ csrf_token() }}",
                            start_date: startDate,
                            end_date: $('#dateSelect').val()
                        }
                    });

                    let workingDays = 0;
                    const currentDate = new Date(start);
                    const holidays = Array.isArray(response) ? response : [];

                    while (currentDate <= selectedDate) {
                        const dayOfWeek = currentDate.getDay();
                        const currentDateStr = currentDate.toISOString().split('T')[0];

                        if (dayOfWeek !== 0 && dayOfWeek !== 6 && !holidays.includes(currentDateStr)) {
                            workingDays++;
                        }

                        currentDate.setDate(currentDate.getDate() + 1);
                    }

                    return Promise.resolve(workingDays);
                } catch (error) {
                    console.error('Error calculating working days:', error);
                    return Promise.resolve(0);
                }
            }

            function updateRegistrationId() {
                $('#lineScheduleTable tbody tr').each(function() {
                    const row = $(this);
                    const factory = $('#factorySelect').val();
                    const date = $('#dateSelect').val();
                    const componentType = row.find('select.area-select').val();
                    const line = row.find('select.line-select').val();
                    const styleNo = row.find('select.style-no-select').val();
                    const buyerShortName = row.find('select.buyer-select').val();
                    const startDate = row.find('.start-date').val();
                    const currentIdReg = row.find('.id-reg').text().trim();

                    if (!currentIdReg || currentIdReg === '' || currentIdReg.startsWith('F1-XX-XXXX')) {
                        if (componentType && ((componentType === 'COMPONENT' && line === 'AREA') ||
                                (componentType === 'ASSEMBLY' && line !== 'AREA')) && styleNo) {
                            const idReg = generateRegistrationId(factory, componentType, line, styleNo,
                                startDate);
                            row.find('.id-reg').text(idReg);
                            const item = scheduleData.find(item => item.ID_REGISTRASI === row.find(
                                '.id-reg').text());
                            if (item) {
                                item.ID_REGISTRASI = idReg;
                                item.COMPONENT = componentType;
                                item.LINE = line;
                                item.STYLE_NO = styleNo;
                                item.BUYER_SHORT_NAME = buyerShortName;
                                item.START_DATE = startDate;
                            }
                        }
                    }
                    if (startDate) {
                        calculateWorkingDays(startDate).then(workingDays => {
                            row.find('td[data-column="WORKING_DAY"]').text(workingDays);
                            const item = scheduleData.find(item => item.ID_REGISTRASI ===
                                currentIdReg);
                            if (item) {
                                item.WORKING_DAY = workingDays;
                            }
                        }).catch(error => {
                            console.error('Error:', error);
                            row.find('td[data-column="WORKING_DAY"]').text('0');
                        });
                    }
                });
            }

            $('#saveChanges').on('click', async function() {
                const selectedFactory = $('#factorySelect').val();
                if (!selectedFactory) {
                    Swal.fire({
                        title: 'Peringatan!',
                        text: 'Silakan pilih factory terlebih dahulu!',
                        icon: 'warning',
                        confirmButtonColor: '#3085d6',
                        confirmButtonText: 'OK'
                    });
                    return;
                }

                const rows = [];
                let hasError = false;
                let errorMessages = [];

                if ($('#lineScheduleTable tbody tr').length === 0) {
                    Swal.fire({
                        title: 'Peringatan!',
                        text: 'Belum ada data yang bisa disimpan. Silakan tambahkan minimal satu baris data.',
                        icon: 'warning',
                        confirmButtonColor: '#3085d6',
                        confirmButtonText: 'OK'
                    });
                    return;
                }

                for (const row of $('#lineScheduleTable tbody tr')) {
                    const $row = $(row);
                    const rowNum = $row.index() + 1;
                    const idReg = $row.find('.id-reg').text().trim();
                    const line = $row.find('select.line-select').val();
                    const styleNo = $row.find('select.style-no-select').val();
                    const buyerShortName = $row.find('select.buyer-select').val();
                    const numOfMP = $row.find('td[data-column="NUMBER_OF_MP"]').text().trim();
                    const startDate = $row.find('.start-date').val();

                    const workingDay = await calculateWorkingDays(startDate);

                    let emptyFields = [];
                    let invalidFields = [];

                    if (!idReg) emptyFields.push('ID Registrasi');
                    if (!line) emptyFields.push('Line');
                    if (!styleNo) emptyFields.push('Style No');
                    if (!buyerShortName) emptyFields.push('Buyer');
                    if (!startDate) emptyFields.push('Tanggal Mulai');
                    if (!numOfMP) {
                        emptyFields.push('Jumlah MP');
                    } else {
                        if (!/^\d+$/.test(numOfMP)) {
                            invalidFields.push('Jumlah MP harus berupa angka');
                            $row.find('td[data-column="NUMBER_OF_MP"]').addClass('error-field');
                        }
                    }

                    if (emptyFields.length > 0 || invalidFields.length > 0) {
                        hasError = true;
                        $row.addClass('error');

                        let errorMessage = `Baris ${rowNum}: `;
                        if (emptyFields.length > 0) {
                            errorMessage +=
                                `${emptyFields.join(', ')} ${emptyFields.length > 1 ? 'wajib diisi' : 'wajib diisi'}`;
                        }
                        if (invalidFields.length > 0) {
                            if (emptyFields.length > 0) errorMessage += ' dan ';
                            errorMessage += invalidFields.join(', ');
                        }
                        errorMessages.push(errorMessage);
                    } else {
                        rows.push({
                            ID_REGISTRASI: idReg,
                            DATE: $('#dateSelect').val(),
                            FACTORY: $('#factorySelect').val(),
                            LINE: line,
                            BUYER_SHORT_NAME: buyerShortName,
                            STYLE_NO: styleNo,
                            START_DATE: startDate,
                            NUMBER_OF_MP: numOfMP,
                            WORKING_DAY: workingDay
                        });
                    }
                }

                if (hasError) {
                    Swal.fire({
                        title: 'Data Tidak Lengkap!',
                        html: '<div style="text-align: left; margin-top: 10px;">' +
                            '<p>Mohon periksa kesalahan berikut:</p>' +
                            '<ul style="list-style-type: disc; margin-left: 20px;">' +
                            errorMessages.map(msg => `<li>${msg}</li>`).join('') +
                            '</ul></div>',
                        icon: 'error',
                        confirmButtonColor: '#3085d6',
                        confirmButtonText: 'Mengerti',
                        width: '600px'
                    });
                    return;
                }

                if (rows.length > 0) {
                    try {
                        const response = await $.post("{{ route('line-schedule.saveChanges') }}", {
                            _token: "{{ csrf_token() }}",
                            data: rows
                        });

                        Swal.fire({
                            title: 'Berhasil!',
                            text: 'Data telah berhasil disimpan!',
                            icon: 'success',
                            confirmButtonColor: '#3085d6',
                            confirmButtonText: 'OK'
                        });
                    } catch (error) {
                        Swal.fire({
                            title: 'Gagal!',
                            text: error.responseJSON?.message ||
                                'Terjadi kesalahan saat menyimpan data.',
                            icon: 'error',
                            confirmButtonColor: '#3085d6',
                            confirmButtonText: 'OK'
                        });
                        console.log(error.responseJSON?.errors);
                    }
                }
            });
