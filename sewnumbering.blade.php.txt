                    ['PROCESS_NAME_IND = Patrun body depan posisi pocket'],
                    ['MACHINE_CODE = S']
                ], {
                    origin: 'M1'
                });

                // Buat workbook
                const workbook = XLSX.utils.book_new();
                XLSX.utils.book_append_sheet(workbook, worksheet, 'Sew Num Process Format');

                // Konversi workbook ke Blob
                const wbout = XLSX.write(workbook, {
                    bookType: 'xlsx',
                    type: 'array'
                });
                const blob = new Blob([wbout], {
                    type: 'application/octet-stream'
                });

                // Buat link untuk mengunduh file
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = 'sew_num_process_format.xlsx';
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
                URL.revokeObjectURL(url);
            });
        });

            function showAutocompleteList() {
                var input = $('#styleInput').val().toLowerCase();
                var selectedBuyer = $('#buyerSelect').val(); // Ambil buyer yang dipilih

                // Jika tidak ada buyer yang dipilih, kosongkan input style dan hidden input, serta jangan tampilkan daftar style
                if (!selectedBuyer) {
                    $('#styleInput').val(''); // Kosongkan input style
                    $('#styleSelect').val(''); // Kosongkan hidden input style
                    $('.autocomplete-list').remove(); // Hapus daftar autocomplete Style jika ada
                    return; // Jangan tampilkan daftar autocomplete Style
                }

                // Jika buyer valid, lanjutkan untuk menampilkan daftar autocomplete style
                var matchedStyles = styles.filter(function(style) {
                    return style.STYLE_NO.toLowerCase().indexOf(input) !== -1 &&
                        style.BUYER_SHORT_NAME === selectedBuyer; // Filter style sesuai buyer
                });

                // Bersihkan list sebelumnya
                $('.autocomplete-list').remove();

                if (matchedStyles.length > 0) {
                    var autocompleteList = $('<ul class="autocomplete-list"></ul>').css({
                        'position': 'absolute',
                        'z-index': '1000',
                        'background-color': '#fff',
                        'border': '1px solid #ccc',
                        'width': $('#styleInput').outerWidth(),
                        'max-height': '150px',
                        'overflow-y': 'auto'
                    });

                    matchedStyles.forEach(function(style) {
                        var listItem = $('<li></li>').text(style.STYLE_NO).css({
                            'padding': '5px',
                            'cursor': 'pointer'

                            $('#styleSelect').change(); // Trigger change event untuk reload tabel

                    });

                }

            function showBuyerAutocompleteList() {
                var input = $('#buyerInput').val().toLowerCase();

                // Buat Set untuk menyimpan buyer yang unik
                var uniqueBuyers = new Set();

                // Filter berdasarkan input dan tambahkan ke Set untuk menghilangkan duplikasi
                var matchedBuyers = styles.filter(function(style) {
                    if (style.BUYER_SHORT_NAME.toLowerCase().indexOf(input) !== -1) {
                        if (!uniqueBuyers.has(style.BUYER_SHORT_NAME)) {
                            uniqueBuyers.add(style.BUYER_SHORT_NAME); // Tambahkan ke Set
                            return true; // Masukkan ke dalam matchedBuyers
                        }
                    }
                    return false;
                });

                // Bersihkan list sebelumnya
                $('.autocomplete-list-buyer').remove();

                if (matchedBuyers.length > 0) {
                    var autocompleteListBuyer = $('<ul class="autocomplete-list-buyer"></ul>').css({
                        'position': 'absolute',
                        'z-index': '1000',
                        'background-color': '#fff',
                        'border': '1px solid #ccc',
                        'width': $('#buyerInput').outerWidth(),
                        'max-height': '150px',
                        'overflow-y': 'auto'
                    });

                    matchedBuyers.forEach(function(buyer) {
                        var listItem = $('<li></li>').text(buyer.BUYER_SHORT_NAME).css({
                            'padding': '5px',
                            'cursor': 'pointer'
                        }).on('click', function() {

                        autocompleteListBuyer.append(listItem);
                    });

                    $('#buyerInput').after(autocompleteListBuyer); // Tampilkan daftar di bawah input
                }
            }

            // Handle save changes
            $('#saveChanges').click(function() {
                changes = []; // Reset changes array
                var validationPassed = true; // Flag untuk memeriksa apakah validasi lulus
                var processNumbers = {}; // Object untuk melacak kemunculan nomor proses

                $('#numProcessTable tbody tr').each(function() {
                    var row = table.row(this);
                    var data = row.data();

                    if (row.node()) {
                        // Reset pesan kesalahan sebelumnya
                        $(row.node()).find('.error-message').remove();

                        var noProcess = $(row.node()).find('td').eq(3).text().trim();
                        var category = $(row.node()).find('td').eq(4).find('select').val();
                        var subCategory = $(row.node()).find('td').eq(5).text().trim();
                        var smvProcGsd = $(row.node()).find('td').eq(6).text().trim();
                        var smvProcEst = $(row.node()).find('td').eq(7).text().trim();
                        var processNameEng = $(row.node()).find('td').eq(8).text().trim();
                        var processNameInd = $(row.node()).find('td').eq(9).text().trim();
                        var machineCode = $(row.node()).find('td').eq(10).text().trim();

                        // Validasi Machine Code (wajib diisi)
                        if (!machineCode) {
                            validationPassed = false;
                            var errorElement = $(
                                '<div class="error-message" style="color: red;">Machine Code is required</div>'
                            );
                            $(row.node()).find('td').eq(10).append(errorElement);
                        }

                        // Validasi Process Number (harus angka)
                        if (isNaN(noProcess) || noProcess === '') {
                            validationPassed = false;
                            var errorElement = $(
                                '<div class="error-message" style="color: red;">This value must be a number</div>'
                            );
                            $(row.node()).find('td').eq(3).append(errorElement);
                        } else if (processNumbers[noProcess]) {
                            // Cek apakah nomor proses sudah ada
                            validationPassed = false;
                            var errorElement = $(
                                '<div class="error-message" style="color: red;">Duplicate Process Number</div>'
                            );
                            $(row.node()).find('td').eq(3).append(errorElement);
                        } else {
                            // Simpan nomor proses ke object untuk pengecekan duplikat
                            processNumbers[noProcess] = true;
                        }

                        // Validasi SMV Proc GSD (boleh kosong, tapi jika diisi harus angka)
                        if (smvProcGsd !== '' && isNaN(smvProcGsd)) {
                            validationPassed = false;
                            var errorElement = $(
                                '<div class="error-message" style="color: red;">If filled, this value must be a number</div>'
                            );
                            $(row.node()).find('td').eq(6).append(errorElement);
                        }

                        // Validasi SMV Proc EST (harus angka)
                        if (isNaN(smvProcEst) || smvProcEst === '') {
                            validationPassed = false;
                            var errorElement = $(
                                '<div class="error-message" style="color: red;">This value must be a number</div>'
                            );
                            $(row.node()).find('td').eq(7).append(errorElement);
                        }

                        // Validasi Process Name IND (wajib diisi)
                        if (!processNameInd) {
                            validationPassed = false;
                            var errorElement = $(
                                '<div class="error-message" style="color: red;">Process Name IND is required</div>'
                            );
                            $(row.node()).find('td').eq(9).append(errorElement);
                        }

                        // Tambahkan event listener untuk menghapus pesan kesalahan ketika kolom difokuskan
                        $(row.node()).find('td').on('click', function() {
                            $(this).find('.error-message').remove();
                            $(this).attr('contenteditable', 'true')
                                .focus(); // Buat kolom langsung bisa di-edit dan fokus
                        });

                        // Jika validasi lulus, simpan data ke dalam array changes
                        var updatedData = {
                            'style_no': data.STYLE_NO,
                            'buyer_short_name': data.BUYER_SHORT_NAME,
                            'original_no_process': data.NO_PROCESS,
                            'no_process': noProcess,
                            'category': category, // New Column
                            'sub_category': subCategory, // New Column
                            'smv_proc_gsd': smvProcGsd,
                            'smv_proc_est': smvProcEst,
                            'process_name_eng': processNameEng || null, // Bisa null
                            'process_name_ind': processNameInd || null, // Bisa null,
                            'machine_code': machineCode,
                            'action': data.action ? data.action : 'edit'
                        };

                        changes.push(updatedData); // Add the updated data to the changes array
                    }
                });

                if (!validationPassed) {
                    return; // Hentikan proses jika validasi gagal
                }

                if (changes.length === 0) {
                    Swal.fire('No Changes', 'There are no changes to save.', 'info');
                    return;
                }

                // AJAX request untuk menyimpan perubahan ke server
                $.ajax({
                    url: "{{ route('numprocess.bulkSave') }}",
                    type: "POST",
                    data: {
                        changes: changes,
                        _token: '{{ csrf_token() }}'
                    },
                    success: function(response) {
                        Swal.fire({
                            icon: 'success',
                            title: 'Success',
                            text: 'All changes have been saved successfully!'
                        });
                        table.ajax.reload();
                        changes = []; // Clear the changes array

                        // Kembalikan tombol Edit ke mode awal setelah save
                        isEditable = false; // Nonaktifkan mode editable
                        $('#editToggle').text('Edit'); // Ubah teks tombol kembali ke "Edit"
                        $('#editToggle').removeClass('btn-outline-secondary delete').addClass(
                            'btn-outline-secondary edit'
                        ); // Ubah warna tombol kembali menjadi biru
                        toggleEditableMode(isEditable); // Nonaktifkan mode editable
                    },
                    error: function(xhr) {
                        Swal.fire({
                            icon: 'error',
                            title: 'Error',
                            text: xhr.responseJSON
                                .error // Menampilkan pesan error yang dikirim dari backend
                        });
                    }
                });
            });

            // Handle form submission for importing Excel
            $('#upload-form').on('submit', function(e) {
                e.preventDefault(); // Mencegah form submission default

                var fileInput = document.getElementById('file');
                var file = fileInput.files[0];
                if (!file) {
                    Swal.fire({
                        icon: 'error',
                        title: 'Error',
                        text: 'Please select an Excel file first.'
                    });
                    return;
                }

                // Tampilkan loading
                Swal.fire({
                    title: 'Importing data...',
                    html: 'Please wait while we process your file',
                    allowOutsideClick: false,
                    didOpen: () => {
                        Swal.showLoading();
                    }
                });

                var reader = new FileReader();
                reader.onload = function(e) {
                    var data = new Uint8Array(e.target.result);
                    var workbook = XLSX.read(data, {
                        type: 'array'
                    });

                    // Ambil sheet pertama
                    var firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                    var jsonData = XLSX.utils.sheet_to_json(firstSheet, {
                        header: 1
                    });

                    var importData = [];

                    // Convert JSON data to the expected format
                    for (var i = 1; i < jsonData.length; i++) { // Mulai dari 1 untuk melewati header
                        var row = jsonData[i];
                        if (row.length > 0) {
                            importData.push({
                                style_no: row[0],
                                buyer_short_name: row[1],
                                no_process: row[2],
                                category: row[3],
                                sub_category: row[4],
                                smv_proc_gsd: row[5],
                                smv_proc_est: row[6],
                                process_name_eng: row[7],
                                process_name_ind: row[8],
                                machine_code: row[9]
                            });
                        }
                    }

                    // Fungsi untuk melakukan import
                    function performImport(forceImport = false) {
                        $.ajax({
                            url: "{{ route('numprocess.import') }}",
                            type: "POST",
                            data: {
                                data: importData,
                                force_import: forceImport,
                                _token: '{{ csrf_token() }}'
                            },
                            success: function(response) {
                                Swal.close(); // Tutup loading
                                if (response.success) {
                                    let message = 'Data has been imported successfully!';
                                    if (response.overwritten_processes > 0) {
                                        message +=
                                            `\n${response.overwritten_processes} processes were overwritten.`;
                                    }
                                    Swal.fire({
                                        icon: 'success',
                                        title: 'Success',
                                        text: message
                                    });
                                    table.ajax.reload();
                                }
                            },
                            error: function(xhr) {
                                Swal.close(); // Tutup loading
                                if (xhr.status === 409) {
                                    // Ada proses yang sudah ada, tampilkan konfirmasi
                                    const response = JSON.parse(xhr.responseText);
                                    let existingProcessList = response.existing_processes
                                        .map(p =>
                                            `<tr>
                                            <td>${p.style_no}</td>
                                            <td>${p.no_process}</td>
                                            <td>${p.current_data.process_name_eng}</td>
                                            <td>${p.current_data.smv_proc_gsd}</td>
                                            <td>${p.current_data.smv_proc_est}</td>
                                        </tr>`
                                        ).join('');

                                    Swal.fire({
                                        title: 'Warning!',
                                        html: `
                                            <p>${response.message}</p>
                                            <div style="max-height: 300px; overflow-y: auto;">
                                                <table class="table table-bordered table-sm">
                                                    <thead>
                                                        <tr>
                                                            <th>Style</th>
                                                            <th>Process No</th>
                                                            <th>Process Name</th>
                                                            <th>SMV GSD</th>
                                                            <th>SMV EST</th>
                                                        </tr>
                                                    </thead>
                                                    <tbody>
                                                        ${existingProcessList}
                                                    </tbody>
                                                </table>
                                            </div>
                                        `,
                                        icon: 'warning',
                                        showCancelButton: true,
                                        confirmButtonText: 'Yes, overwrite them',
                                        cancelButtonText: 'No, cancel import',
                                        confirmButtonColor: '#d33',
                                        width: '800px'
                                    }).then((result) => {
                                        if (result.isConfirmed) {
                                            performImport(true);
                                        }
                                    });
                                } else {
                                    // Error lainnya
                                    Swal.fire({
                                        icon: 'error',
                                        title: 'Error',
                                        text: 'Failed to import data. ' +
                                            (xhr.responseJSON?.error || xhr
                                                .responseJSON?.message ||
                                                'Unknown error occurred.')
                                    });
                                }
                            }
                        });
                    }

                    // Mulai import
                    performImport(false);
                };
                reader.readAsArrayBuffer(file);
            });

            // Validasi file sebelum upload
            $('#file').change(function() {
                const file = this.files[0];
                const maxSize = 5 * 1024 * 1024; // 5MB
                const allowedTypes = ['application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'];

                if (file.size > maxSize) {
                    Swal.fire({
                        icon: 'error',
                        title: 'Error',
                        text: 'Ukuran file tidak boleh lebih dari 5MB'
                    });
                    this.value = '';
                    return;
                }

                if (!allowedTypes.includes(file.type)) {
                    Swal.fire({
                        icon: 'error',
                        title: 'Error',
                        text: 'Format file harus .xlsx'
                    });
                    this.value = '';
                    return;
                }
            });
